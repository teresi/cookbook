\section[Pimento Dram]{Pimento Dram, or\\Allspice Liqueur}\index{allspice!Pimento~Dram}

\begin{recipestats}[
        servings=24~\fluidounce,
        preptime=15~\minute,
        inactivetime=12~day,
        original=\citefield{allspiceDram}{shorttitle}~\cite{allspiceDram},
    ]
\end{recipestats}

\begin{recipeabstract}
    A syrup of rum and allspice (aka pimento).
    The name allspice is given as it tastes similar to cinnamon, nutmeg, and cloves.
    The liqueur is used in small quantities in drinks such as Navy Grog and Planter's Punch.
\end{recipeabstract}

\begin{ingredientcolumns}
    \begin{ingredientblock}
        \ingredient[1~\onehalf][C]{Jamaican Rum}\\
        \ingredient[\onethird][C]{allspice}\\
        \ingredient[1][stick]{cinnamon}\\
        \ingredient[1][pod]{star anise}
    \end{ingredientblock}
    \begin{ingredientblock}
        \ingredient[1][C]{demerara sugar}\\
        \ingredient[1][C]{water}
    \end{ingredientblock}
\end{ingredientcolumns}

\ragmarpar{Crack open the berries, don't finely grind.}
\begin{preparation}
\item Roughly crush the allspice.
\item Add allspice and rum to a mason jar.
\item Macerate for $4..5$ $day$, shaking each day.
\item Add cinnamon and anise.
\item Macerate for $7$ $day$.
\item Strain through a coffee filter into a large mason jar.
\item Add sugar and water, let dissolve, refridgerate.
\item Wait 2 $day$ before using.
\end{preparation}

\recipeend%
