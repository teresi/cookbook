\section[Agua de Jamaica]{Agua de Jamaica\\or, Hibiscus Tea}\index{ingredient!Hibiscus}

\begin{recipestats}[
        servings=1~\gallon,
        preptime=10~\minute,
        bakingtime=45~\minute,
        inactivetime=15~\minute,
        original=\cite{aguaDeJamaica},
        source=Mike \& Jane,
    ]
\end{recipestats}

\begin{recipeabstract}
    A refreshing ruby red beverage.
    Serve ice cold.
    The Hibiscus provides a tart base that is spiced and sweetened.
    We like to add lime to round it out.
\end{recipeabstract}

\begin{ingredientcolumns}
    \begin{ingredientblock}
        \ingredient[6][C]{water}\\
        \ingredient[1~\onehalf][C]{sugar}\\
        \ingredient[\onehalf][C]{piloncillo}\\
        \ingredient[2][]{lime}
    \end{ingredientblock}
    \begin{ingredientblock}
        \ingredient[2][C]{hibiscus}\\
        \ingredient[1][stick]{canela}\\
        \ingredient[1][berry]{allspice}\\
        \ingredient[1][dash]{nutmeg}\\
        \ingredient[1][dash]{canela}
    \end{ingredientblock}
\end{ingredientcolumns}
\pagebreak[1]

\begin{preparation}
\item Reserve two small pots and a $\onehalf~\gallon$ pitcher.
\item Begin to boil the water.
\item Measure the flowers and spices, set aside.
\item Measure the sugar, set aside.
    \ragmarpar{Shave the piloncillo with a knife.}
\item Once boiling, add spices, reduce to simmer.
\item Simmer $\approx[30..45]~\minute$.
\item Take off the heat, let steep $15~\minute$.
\item Strain juice into a pot, press the flowers to release the juice.
\item Stir in the sugar until fully dissolved.
    \ragmarpar{Yields $\approx 5~\onehalf$ $C$ concentrate.}
\item Split half of the juice into the pitcher, fill with ice and water.
\item Squeeze the lime juice into the pitcher and stir.
\item Set aside the remaining concentrate for another batch.
\end{preparation}
